系统配置工具

系统配置工具是Linux系统管理和配置的核心工具集，包括系统日志、时钟管理、终端工具等。本章将详细介绍这些工具的编译和安装。

🎯 工具概述

工具分类

系统配置工具主要包括：

  • 系统日志工具：sysklogd, syslog-ng
  • 时钟管理工具：util-linux (hwclock)
  • 终端工具：ncurses, readline
  • 文档工具：man-db, texinfo
  • 其他系统工具：kbd, console-data

📝 Sysklogd工具

编译Sysklogd

# 切换到lfs用户
su - lfs

# 创建构建目录
mkdir -pv $LFS/sources/system_tools
cd $LFS/sources/system_tools

# 解压Sysklogd源码
tar -xf $LFS/sources/sysklogd-1.5.1.tar.gz
cd sysklogd-1.5.1

# 配置Sysklogd
sed -i '/Error loading kernel symbols/{n;n;d}' ksym_mod.c
sed -i 's/union wait/int/' syslogd.c

# 编译
make $LFS_MAKEFLAGS

# 安装
make BINDIR=/sbin install

# 创建配置文件
cat > /etc/syslog.conf << "EOF"
# Begin /etc/syslog.conf

auth,authpriv.* -/var/log/auth.log
*.*;auth,authpriv.none -/var/log/sys.log
daemon.* -/var/log/daemon.log
kern.* -/var/log/kern.log
mail.* -/var/log/mail.log
user.* -/var/log/user.log
*.emerg *

# End /etc/syslog.conf
EOF

🖥️ Ncurses库

编译Ncurses

# 返回源码目录
cd $LFS/sources/system_tools

# 解压Ncurses源码
tar -xf $LFS/sources/ncurses-6.4.tar.gz
cd ncurses-6.4

# 配置Ncurses
./configure --prefix=/usr \
            --mandir=/usr/share/man \
            --with-shared \
            --without-debug \
            --without-normal \
            --with-cxx-shared \
            --enable-pc-files \
            --enable-widec \
            --with-pkg-config-libdir=/usr/lib/pkgconfig

# 编译
make $LFS_MAKEFLAGS

# 安装
make DESTDIR=$LFS install

# 修复库文件
mv -v $LFS/usr/lib/libncursesw.so.6* /usr/lib
ln -sfv ../../lib/$(readlink $LFS/usr/lib/libncursesw.so) /usr/lib/libncursesw.so

# 创建兼容性符号链接
for lib in ncurses form panel menu ; do
    rm -vf /usr/lib/lib${lib}.so
    echo "INPUT(-l${lib}w)" > /usr/lib/lib${lib}.so
    ln -sfv ${lib}w.pc /usr/lib/pkgconfig/${lib}.pc
done

# 安装文档
rm -vf /usr/lib/libcursesw.so
echo "INPUT(-lncursesw)" > /usr/lib/libcursesw.so
ln -sfv libncurses.so /usr/lib/libcurses.so

# 清理
rm -rf $LFS/usr/share/man

📖 Readline库

编译Readline

📋 构建脚本

自动化构建脚本

# 创建系统工具构建脚本
cat > $LFS/build_system_tools.sh << 'EOF'
#!/bin/bash
# LFS系统工具构建脚本

set -e

# 工具列表
tools=(
    "sysklogd-1.5.1:sysklogd"
    "ncurses-6.4:ncurses"
    "readline-8.2:readline"
    "man-db-2.11.2:man-db"
    "kbd-2.5.1:kbd"
)

total_tools=${#tools[@]}
completed=0

for tool_info in "${tools[@]}"; do
    IFS=':' read -r package_name tool_name <<< "$tool_info"

    echo "=== 构建 $tool_name ($((completed + 1))/$total_tools) ==="

    # 检查源码
    if [ ! -f "$LFS/sources/$package_name.tar.xz" ] && [ ! -f "$LFS/sources/$package_name.tar.gz" ]; then
        echo "错误: $package_name 源码不存在"
        exit 1
    fi

    cd $LFS/sources/system_tools

    # 解压源码
    if [ -f "$LFS/sources/$package_name.tar.xz" ]; then
        tar -xf "$LFS/sources/$package_name.tar.xz"
    else
        tar -xf "$LFS/sources/$package_name.tar.gz"
    fi

    cd $package_name

    # 特殊构建流程
    case $tool_name in
        sysklogd)
            sed -i '/Error loading kernel symbols/{n;n;d}' ksym_mod.c
            sed -i 's/union wait/int/' syslogd.c
            make $LFS_MAKEFLAGS
            make BINDIR=/sbin install
            cat > /etc/syslog.conf << "EOF"
auth,authpriv.* -/var/log/auth.log
*.*;auth,authpriv.none -/var/log/sys.log
daemon.* -/var/log/daemon.log
kern.* -/var/log/kern.log
mail.* -/var/log/mail.log
user.* -/var/log/user.log
*.emerg *
EOF
            ;;

        ncurses)
            ./configure --prefix=/usr \
                        --mandir=/usr/share/man \
                        --with-shared \
                        --without-debug \
                        --without-normal \
                        --with-cxx-shared \
                        --enable-pc-files \
                        --enable-widec \
                        --with-pkg-config-libdir=/usr/lib/pkgconfig
            make $LFS_MAKEFLAGS
            make DESTDIR=$LFS install
            mv -v $LFS/usr/lib/libncursesw.so.6* /usr/lib
            ln -sfv ../../lib/$(readlink $LFS/usr/lib/libncursesw.so) /usr/lib/libncursesw.so
            for lib in ncurses form panel menu ; do
                rm -vf /usr/lib/lib${lib}.so
                echo "INPUT(-l${lib}w)" > /usr/lib/lib${lib}.so
                ln -sfv ${lib}w.pc /usr/lib/pkgconfig/${lib}.pc
            done
            rm -vf /usr/lib/libcursesw.so
            echo "INPUT(-lncursesw)" > /usr/lib/libcursesw.so
            ln -sfv libncurses.so /usr/lib/libcurses.so
            rm -rf $LFS/usr/share/man
            ;;

        readline)
            ./configure --prefix=/usr \
                        --disable-static \
                        --with-curses \
                        --docdir=/usr/share/doc/readline-8.2
            make $LFS_MAKEFLAGS SHLIB_LIBS="-lncursesw"
            make SHLIB_LIBS="-lncursesw" install
            install -v -m644 doc/*.{ps,pdf,html,dvi} /usr/share/doc/readline-8.2
            ;;

        man-db)
            ./configure --prefix=/usr \
                        --docdir=/usr/share/doc/man-db-2.11.2 \
                        --sysconfdir=/etc \
                        --disable-setuid \
                        --enable-cache-owner=bin \
                        --with-browser=/usr/bin/lynx \
                        --with-vgrind=/usr/bin/vgrind \
                        --with-grap=/usr/bin/grap \
                        --with-systemdtmpfilesdir= \
                        --with-systemdsystemunitdir=
            make $LFS_MAKEFLAGS
            make install
            cat > /etc/man_db.conf << "EOF"
MANPATH_MAP /bin /usr/share/man
MANPATH_MAP /usr/bin /usr/share/man
MANPATH_MAP /sbin /usr/share/man
MANPATH_MAP /usr/sbin /usr/share/man
MANPATH_MAP /usr/local/bin /usr/local/share/man
MANPATH_MAP /usr/local/sbin /usr/local/share/man
MANPATH_MAP /usr/X11R6/bin /usr/X11R6/share/man
EOF
            ;;

        kbd)
            patch -Np1 -i ../kbd-2.5.1-backspace-1.patch
            ./configure --prefix=/usr --disable-vlock
            make $LFS_MAKEFLAGS
            make install
            mkdir -v /usr/share/doc/kbd-2.5.1
            cp -R -v docs/doc/* /usr/share/doc/kbd-2.5.1
            ;;
    esac

    # 验证安装
    echo "验证 $tool_name 安装..."
    case $tool_name in
        sysklogd)
            [ -x /sbin/syslogd ] && echo "✓ sysklogd 安装成功" || echo "✗ sysklogd 安装失败"
            ;;
        ncurses)
            [ -f /usr/lib/libncursesw.so ] && echo "✓ ncurses 安装成功" || echo "✗ ncurses 安装失败"
            ;;
        readline)
            [ -f /usr/lib/libreadline.so ] && echo "✓ readline 安装成功" || echo "✗ readline 安装失败"
            ;;
        man-db)
            [ -x /usr/bin/man ] && echo "✓ man-db 安装成功" || echo "✗ man-db 安装失败"
            ;;
        kbd)
            [ -x /usr/bin/loadkeys ] && echo "✓ kbd 安装成功" || echo "✗ kbd 安装失败"
            ;;
    esac

    completed=$((completed + 1))
    echo "进度: $completed/$total_tools 完成"
    echo ""

    # 清理构建目录
    cd $LFS/sources/system_tools
    rm -rf $package_name
done

echo "=== 所有系统工具构建完成 ==="
EOF

chmod +x $LFS/build_system_tools.sh

🧪 功能验证

工具可用性测试

# 创建验证脚本
cat > $LFS/verify_system_tools.sh << 'EOF'
#!/bin/bash
# 系统工具验证脚本

echo "=== LFS系统工具验证 ==="

# 定义要验证的工具
tools=(
    "/sbin/syslogd:sysklogd"
    "/usr/lib/libncursesw.so:ncurses"
    "/usr/lib/libreadline.so:readline"
    "/usr/bin/man:man-db"
    "/usr/bin/loadkeys:kbd"
)

passed=0
total=${#tools[@]}

for tool_info in "${tools[@]}"; do
    IFS=':' read -r tool_path tool_name <<< "$tool_info"

    echo -n "检查 $tool_name ($tool_path)... "

    if [ -e "$tool_path" ]; then
        echo "✓ 存在"

        # 基本功能测试
        case $tool_name in
            sysklogd)
                $tool_path --version >/dev/null 2>&1 && echo "  版本检查: ✓" || echo "  版本检查: ✗"
                ;;
            ncurses)
                # 检查是否可以链接
                echo "#include <ncurses.h>" > test_ncurses.c
                echo "int main() { initscr(); endwin(); return 0; }" >> test_ncurses.c
                gcc test_ncurses.c -lncursesw -o test_ncurses >/dev/null 2>&1 && echo "  编译测试: ✓" || echo "  编译测试: ✗"
                rm -f test_ncurses.c test_ncurses
                ;;
            readline)
                # 检查readline头文件
                [ -f /usr/include/readline/readline.h ] && echo "  头文件: ✓" || echo "  头文件: ✗"
                ;;
            man-db)
                man --version >/dev/null 2>&1 && echo "  版本检查: ✓" || echo "  版本检查: ✗"
                ;;
            kbd)
                $tool_path --version >/dev/null 2>&1 && echo "  版本检查: ✓" || echo "  版本检查: ✗"
                ;;
        esac

        passed=$((passed + 1))
    else
        echo "✗ 不存在"
    fi
done

echo ""
echo "=== 验证结果 ==="
echo "通过: $passed/$total"

if [ $passed -eq $total ]; then
    echo "✓ 所有系统工具都已正确安装"
    exit 0
else
    echo "✗ 部分工具安装失败"
    exit 1
fi
EOF

chmod +x $LFS/verify_system_tools.sh

🚨 常见问题

Ncurses编译问题

# 如果Ncurses编译失败：
# 1. 检查依赖
echo "检查gcc版本..."
gcc --version

# 2. 检查配置
echo "检查配置选项..."
cd ncurses-6.4
./configure --help | grep -A5 -B5 shared

# 3. 重新配置
make clean
./configure --prefix=/usr --with-shared --enable-widec

Man-db配置问题

# 如果man-db配置有问题：
# 1. 检查groff是否安装
which groff || echo "groff未安装"

# 2. 检查配置文件
cat /etc/man_db.conf

# 3. 重新生成man数据库
mandb -c

Kbd工具问题

# 如果kbd有问题：
# 1. 检查内核头文件
ls -la /usr/include/linux/

# 2. 检查补丁应用
cd kbd-2.5.1
ls -la *.patch 2>/dev/null || echo "补丁文件不存在"

# 3. 重新应用补丁
patch -Np1 -i ../kbd-2.5.1-backspace-1.patch
